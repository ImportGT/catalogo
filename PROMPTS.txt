def actualizar_todo():
    for conf in CONFIGURACIONES:
        cat = conf["categoria"]
        excel_path = conf["excel"]
        js_path = conf["js"]
        var_name = conf["variable_js"]
        prefijo = conf["prefijo"]
        carpeta = conf["carpeta"]

        print(f"\n--- Procesando categoría: {cat} ---")

        if not os.path.exists(excel_path):
            print(f"⚠️ Aviso: No se encontró el archivo {excel_path}. Se omite esta categoría.")
            continue

        if not os.path.exists(carpeta):
            print(f"⚠️ Aviso: No se encontró la carpeta {carpeta}. Se omite esta categoría.")
            continue

        df = pd.read_excel(excel_path)
        productos_lista = []

        for index, row in df.iterrows():
            col_id = 'NUMERO' if 'NUMERO' in df.columns else 'ID'
            if col_id not in df.columns or pd.isna(row[col_id]):
                continue
                
            prod_id = int(row[col_id])
            
            col_precio = 'PRECIO' if 'PRECIO' in df.columns else 'Precio'
            precio = float(row[col_precio]) if col_precio in df.columns and not pd.isna(row[col_precio]) else 0.0
            
            galeria_items = []
            
            for archivo in os.listdir(carpeta):
                if (archivo.startswith(f"{prefijo}.{prod_id}.") or 
                    archivo.startswith(f"{prefijo}_{prod_id}_") or 
                    archivo.startswith(f"{prefijo}_{prod_id}") or 
                    archivo.startswith(f"{prefijo}.{prod_id}")):
                    
                    ruta_archivo = f"{carpeta}/{archivo}"
                    
                    if archivo.lower().endswith(('.mp4', '.mov', '.webm')):
                        galeria_items.append({"tipo": "video", "url": ruta_archivo})
                    elif archivo.lower().endswith(('.jpg', '.jpeg', '.png', '.webp')):
                        galeria_items.append({"tipo": "imagen", "url": ruta_archivo})
            
            galeria_items = sorted(galeria_items, key=lambda x: x['url'])
            
            imagenes_solas = [item for item in galeria_items if item['tipo'] == 'imagen']
            if imagenes_solas:
                imagen_principal = imagenes_solas[0]['url']
            elif galeria_items:
                imagen_principal = galeria_items[0]['url']
            else:
                imagen_principal = f"{carpeta}/{prefijo}_{prod_id}.0.jpg"

            producto_obj = {
                "id": prod_id,
                "categoria": cat,
                "precio": precio,
                "imagen": imagen_principal
            }
            
            if len(galeria_items) > 1:
                producto_obj["galeria"] = galeria_items
                
            productos_lista.append(producto_obj)

        contenido_js = f"const {var_name} = {json.dumps(productos_lista, indent=4, ensure_ascii=False)};"

        with open(js_path, "w", encoding="utf-8") as f:
            f.write(contenido_js)

        print(f"✅ Archivo {js_path} actualizado con {len(productos_lista)} productos.")

    print("\n🎉 ¡Proceso de actualización completo para todo el catálogo!")

if __name__ == "__main__":
    actualizar_todo()

        const productos = [
            ...(typeof productosAnillos !== 'undefined' ? productosAnillos : []),
            ...(typeof productosAretes !== 'undefined' ? productosAretes : []),

            ...(typeof productosCharmsColgantesDisney !== 'undefined' ? productosCharmsColgantesDisney : []),
            ...(typeof productosCharmsReflexion !== 'undefined' ? productosCharmsReflexion : []),
            ...(typeof productosCharmsMuranos !== 'undefined' ? productosCharmsMuranos : []),
            ...(typeof productosCharmsCadenasSeguridad !== 'undefined' ? productosCharmsCadenasSeguridad : []),
            ...(typeof productosCharmsClipsYTopes !== 'undefined' ? productosCharmsClipsYTopes : []),
            ...(typeof productosCharmsLocket !== 'undefined' ? productosCharmsLocket : []),

        ];

        const nombreLinea = "PANDORA";
        let carrito = JSON.parse(localStorage.getItem('mi_carrito_catalogo')) || {};
        let preciosVisibles = true;

        let lightboxGaleria = [];
        let lightboxIndiceActual = 0;

        function guardarCarrito() {
            localStorage.setItem('mi_carrito_catalogo', JSON.stringify(carrito));
        }

        function limpiarCarrito() {
            if (Object.keys(carrito).length === 0) {
                alert("El carrito ya está vacío.");
                return;
            }
            if (confirm("¿Estás seguro de que deseas vaciar todo el carrito de compras?")) {
                carrito = {};
                localStorage.removeItem('mi_carrito_catalogo');
                renderizarCatalogo();
                actualizarBarraCarrito();
                cerrarModalCarrito();
            }
        }

        function calcularPrecioFinal(precioBase) {
            let margen = parseFloat(document.getElementById('input-ganancia').value) || 0;
            return precioBase * (1 + (margen / 100));
        }

        function renderizarCatalogo() {
            const contenedor = document.getElementById('catalogo-contenido');
            const indiceContenedor = document.getElementById('indice-categorias');
            contenedor.innerHTML = '';
            indiceContenedor.innerHTML = '';

            const categorias = {};
            productos.forEach(p => {
                if (!categorias[p.categoria]) categorias[p.categoria] = [];
                categorias[p.categoria].push(p);
            });

            for (let cat in categorias) {
                let idCat = cat.toLowerCase().replace(/\s+/g, '-');
                indiceContenedor.innerHTML += `<a href="#${idCat}">${cat}</a>`;

                let seccion = document.createElement('div');
                seccion.className = 'category-section';
                seccion.id = idCat;
                seccion.innerHTML = `<div class="category-title">${cat}</div>`;

                let grid = document.createElement('div');
                grid.className = 'products-grid';

                categorias[cat].forEach(prod => {
                    let precioFinal = calcularPrecioFinal(prod.precio);
                    let uniqueKey = prod.id + '-' + prod.categoria;
                    let qty = carrito[uniqueKey] ? carrito[uniqueKey].qty : 0;

                    let card = document.createElement('div');
                    card.className = 'product-card';
                    card.innerHTML = `
                        <img src="${prod.imagen}" alt="Código ${prod.id}" loading="lazy" onclick="abrirGaleria('${prod.categoria}', ${prod.id})" title="Haz clic para ver galería y videos">
                        <div class="product-code">Cód. ${prod.id}</div>
                        <div class="product-price" style="display: ${preciosVisibles ? 'block' : 'none'};">Q${precioFinal.toFixed(2)}</div>
                        <div class="qty-control">
                            <button onclick="cambiarCantidad(${prod.id}, '${prod.categoria}', -1)">-</button>
                            <span class="qty-display" id="qty-${prod.categoria}-${prod.id}">${qty}</span>
                            <button onclick="cambiarCantidad(${prod.id}, '${prod.categoria}', 1)">+</button>
                        </div>
                    `;
                    grid.appendChild(card);
                });

                seccion.appendChild(grid);
                contenedor.appendChild(seccion);
            }
            actualizarBarraCarrito();
        }

        function abrirGaleria(categoria, id) {
            let prod = productos.find(p => p.id === id && p.categoria === categoria);
            if (!prod) return;

            document.getElementById('gallery-modal-title').innerText = `Galería - Cód. ${prod.id} (${prod.categoria})`;
            const container = document.getElementById('gallery-items-container');
            container.innerHTML = '';

            let elementosGaleria = prod.galeria;
            if (!elementosGaleria || elementosGaleria.length === 0) {
                elementosGaleria = [{ tipo: 'imagen', url: prod.imagen }];
            }

            lightboxGaleria = elementosGaleria;

            elementosGaleria.forEach((item, index) => {
                if (item.tipo === 'video') {
                    let videoEl = document.createElement('video');
                    videoEl.src = item.url;
                    videoEl.controls = true;
                    videoEl.preload = 'metadata';
                    videoEl.onclick = function(e) {
                        e.stopPropagation();
                        abrirLightbox(index);
                    };
                    container.appendChild(videoEl);
                } else {
                    let imgEl = document.createElement('img');
                    imgEl.src = item.url;
                    imgEl.onclick = function() {
                        abrirLightbox(index);
                    };
                    container.appendChild(imgEl);
                }
            });

            document.getElementById('modal-galeria').style.display = 'flex';
        }

        function cerrarModalGaleria() { 
            document.getElementById('modal-galeria').style.display = 'none'; 
        }

        function abrirLightbox(indice) {
            lightboxIndiceActual = indice;
            actualizarLightboxMedia();
            document.getElementById('modal-lightbox').style.display = 'flex';
        }

        function cerrarLightbox() {
            document.getElementById('modal-lightbox').style.display = 'none';
            const vid = document.getElementById('lightbox-active-video');
            if (vid) vid.pause();
        }

        function cambiarItemLightbox(direccion) {
            const vid = document.getElementById('lightbox-active-video');
            if (vid) vid.pause();

            lightboxIndiceActual += direccion;
            if (lightboxIndiceActual >= lightboxGaleria.length) {
                lightboxIndiceActual = 0;
            } else if (lightboxIndiceActual < 0) {
                lightboxIndiceActual = lightboxGaleria.length - 1;
            }
            actualizarLightboxMedia();
        }

        function actualizarLightboxMedia() {
            if (lightboxGaleria.length > 0) {
                const mediaContainer = document.getElementById('lightbox-media-container');
                mediaContainer.innerHTML = '';

                let itemActual = lightboxGaleria[lightboxIndiceActual];

                if (itemActual.tipo === 'video') {
                    let videoEl = document.createElement('video');
                    videoEl.id = 'lightbox-active-video';
                    videoEl.src = itemActual.url;
                    videoEl.controls = true;
                    videoEl.autoplay = true;
                    videoEl.className = 'lightbox-video-content';
                    mediaContainer.appendChild(videoEl);
                } else {
                    let imgEl = document.createElement('img');
                    imgEl.src = itemActual.url;
                    imgEl.className = 'lightbox-content';
                    mediaContainer.appendChild(imgEl);
                }

                document.getElementById('lightbox-counter').innerText = `${lightboxIndiceActual + 1} / ${lightboxGaleria.length}`;
                
                let showNav = lightboxGaleria.length > 1;
                document.getElementById('lightbox-btn-prev').style.display = showNav ? 'block' : 'none';
                document.getElementById('lightbox-btn-next').style.display = showNav ? 'block' : 'none';
            }
        }

        let touchStartX = 0;
        let touchEndX = 0;

        const lightboxEl = document.getElementById('modal-lightbox');
        lightboxEl.addEventListener('touchstart', e => {
            touchStartX = e.changedTouches[0].screenX;
        }, false);

        lightboxEl.addEventListener('touchend', e => {
            touchEndX = e.changedTouches[0].screenX;
            manejarDeslizamiento();
        }, false);

        function manejarDeslizamiento() {
            let umbral = 50;
            if (touchEndX < touchStartX - umbral) {
                cambiarItemLightbox(1);
            }
            if (touchEndX > touchStartX + umbral) {
                cambiarItemLightbox(-1);
            }
        }

        function cambiarCantidad(id, categoria, cambio) {
            let uniqueKey = id + '-' + categoria;
            if (!carrito[uniqueKey]) {
                let prod = productos.find(p => p.id === id && p.categoria === categoria);
                if (!prod) return;
                
                let tallaInicial = "Única";
                if (categoria === "Anillos") tallaInicial = "6";
                else if (categoria === "Pulseras") tallaInicial = "17";

                carrito[uniqueKey] = { ...prod, qty: 0, talla: tallaInicial };
            }
            carrito[uniqueKey].qty += cambio;
            if (carrito[uniqueKey].qty <= 0) delete carrito[uniqueKey];
            
            guardarCarrito();

            let display = document.getElementById(`qty-${categoria}-${id}`);
            if (display) display.innerText = carrito[uniqueKey] ? carrito[uniqueKey].qty : 0;
            actualizarBarraCarrito();
        }

        function actualizarBarraCarrito() {
            let totalPiezas = 0, totalPrice = 0;
            for (let key in carrito) {
                totalPiezas += carrito[key].qty;
                totalPrice += calcularPrecioFinal(carrito[key].precio) * carrito[key].qty;
            }
            document.getElementById('cart-summary').innerText = `Piezas: ${totalPiezas} | Total: Q${totalPrice.toFixed(2)}`;
            document.getElementById('cart-modal-total').innerText = `Total: Q${totalPrice.toFixed(2)}`;
        }

        function abrirModalCarrito() {
            const container = document.getElementById('cart-items-container');
            container.innerHTML = '';
            let keys = Object.keys(carrito);
            
            if (keys.length === 0) {
                container.innerHTML = `<div class="empty-cart-msg">Tu carrito está vacío.</div>`;
            } else {
                keys.forEach(key => {
                    let item = carrito[key];
                    let precioItem = calcularPrecioFinal(item.precio);
                    let row = document.createElement('div');
                    row.className = 'cart-item-row';
                    row.innerHTML = `
                        <img src="${item.imagen}" class="cart-item-img">
                        <div class="cart-item-details">
                            <strong>Cód. ${item.id} (${item.categoria})</strong>
                            Q${precioItem.toFixed(2)} c/u
                            <div class="cart-size-select">
                                ${item.categoria === 'Anillos' ? 'Talla:' : (item.categoria === 'Pulseras' ? 'Talla:' : 'Medida/Tipo:')}
                                <select onchange="cambiarTalla('${key}', this.value)">
                                    ${item.categoria === 'Anillos' ? `
                                        <option value="5" ${item.talla=="5"?"selected":""}>5</option>
                                        <option value="6" ${item.talla=="6"?"selected":""}>6</option>
                                        <option value="7" ${item.talla=="7"?"selected":""}>7</option>
                                        <option value="8" ${item.talla=="8"?"selected":""}>8</option>
                                        <option value="9" ${item.talla=="9"?"selected":""}>9</option>
                                    ` : (item.categoria === 'Pulseras' ? `
                                        <option value="17" ${item.talla=="17"?"selected":""}>17</option>
                                        <option value="18" ${item.talla=="18"?"selected":""}>18</option>
                                        <option value="19" ${item.talla=="19"?"selected":""}>19</option>
                                        <option value="20" ${item.talla=="20"?"selected":""}>20</option>
                                        <option value="21" ${item.talla=="21"?"selected":""}>21</option>
                                    ` : `
                                        <option value="Única" ${item.talla=="Única"?"selected":""}>Única</option>
                                    `)}
                                </select>
                            </div>
                        </div>
                        <div class="cart-item-actions">
                            <button onclick="cambiarCantidadModal('${key}', -1)">-</button>
                            <span class="cart-item-qty">${item.qty}</span>
                            <button onclick="cambiarCantidadModal('${key}', 1)">+</button>
                        </div>
                    `;
                    container.appendChild(row);
                });
            }
            document.getElementById('modal-carrito').style.display = 'flex';
        }

        function cerrarModalCarrito() { document.getElementById('modal-carrito').style.display = 'none'; }
        
        function cambiarTalla(key, t) { 
            if (carrito[key]) {
                carrito[key].talla = t; 
                guardarCarrito();
            }
        }
        
        function cambiarCantidadModal(key, c) {
            let item = carrito[key];
            if (item) {
                cambiarCantidad(item.id, item.categoria, c);
                abrirModalCarrito();
            }
        }

        function alternarModoPrecios() {
            preciosVisibles = !preciosVisibles;
            let btn = document.getElementById('btn-precios');
            btn.className = preciosVisibles ? 'toggle-btn' : 'toggle-btn active';
            btn.innerText = preciosVisibles ? '👁️ Precios: VISIBLES' : '👁️ Precios: OCULTOS';
            renderizarCatalogo();
        }

        function actualizarEnlaceCompartible() {
            let w = document.getElementById('input-whatsapp').value;
            let g = document.getElementById('input-ganancia').value;
            document.getElementById('link-generado').value = `${window.location.origin}${window.location.pathname}?w=${w}&g=${g}`;
        }

        function copiarEnlace() {
            let input = document.getElementById('link-generado');
            input.select();
            document.execCommand('copy');
            alert('¡Enlace copiado al portapapeles!');
        }

        function descargarResumenPDF() {
            const { jsPDF } = window.jspdf;
            const doc = new jsPDF();
            doc.setFont("helvetica", "bold");
            doc.setFontSize(16);
            doc.text("RESUMEN DE PEDIDO - " + nombreLinea, 14, 20);
            doc.setFontSize(10);
            doc.setFont("helvetica", "normal");
            doc.text("Fecha: " + new Date().toLocaleDateString(), 14, 28);
            
            let y = 40;
            doc.setFont("helvetica", "bold");
            doc.text("Cód.", 14, y);
            doc.text("Categoría", 35, y);
            doc.text("Talla/Detalle", 70, y);
            doc.text("Cant.", 100, y);
            doc.text("Subtotal", 125, y);
            y += 6; doc.line(14, y, 190, y); y += 8;
            
            doc.setFont("helvetica", "normal");
            let totalGeneral = 0;
            for (let key in carrito) {
                let item = carrito[key];
                let sub = calcularPrecioFinal(item.precio) * item.qty;
                totalGeneral += sub;
                doc.text(`Cód. ${item.id}`, 14, y);
                doc.text(item.categoria, 35, y);
                doc.text(String(item.talla), 70, y);
                doc.text(String(item.qty), 100, y);
                doc.text(`Q${sub.toFixed(2)}`, 125, y);
                y += 8;
                if (y > 270) { doc.addPage(); y = 20; }
            }
            y += 5; doc.line(14, y, 190, y); y += 10;
            doc.setFont("helvetica", "bold");
            doc.text(`TOTAL GENERAL: Q${totalGeneral.toFixed(2)}`, 14, y);
            doc.save("Resumen_Pedido.pdf");
        }

        window.onload = function() {
            const urlParams = new URLSearchParams(window.location.search);
            if (urlParams.has('w')) document.getElementById('input-whatsapp').value = urlParams.get('w');
            if (urlParams.has('g')) document.getElementById('input-ganancia').value = urlParams.get('g');
            actualizarEnlaceCompartible();
            renderizarCatalogo();
        };
